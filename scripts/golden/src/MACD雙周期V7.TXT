var bool is30m_rising  = false
var int  rise_count    = 0
bool is30m_green       = histFast > 0
bool is30m_first_drop  = false

if is30m_chart and is30m_green
    if not is30m_rising
        is30m_rising := true
        rise_count   := 1
    else
        if histFast > histFast[1]
            rise_count += 1
        else if histFast < histFast[1]
            if rise_count >= firstTurnMinBars
                is30m_first_drop := true
            rise_count   := 0
            is30m_rising := false
else
    is30m_rising     := false
    rise_count       := 0
    is30m_first_drop := false

bool is30m_first_rise  = false

if is30m_chart and is30m_red
    if not is30m_falling
        is30m_falling := true
        fall_count    := 1
    else
        // 紅柱負值變大 → 柱體增加（趨勢延續）
        if histFast < histFast[1]
            fall_count += 1
        // 紅柱負值變小 → 柱體減少（首次回升）
        else if histFast > histFast[1]
            if fall_count >= firstTurnMinBars
                is30m_first_rise := true
            fall_count    := 0
            is30m_falling := false
else
    is30m_falling    := false
    fall_count       := 0
    is30m_first_rise := false

// ===== V7.1：15分短週期MACD弱動能停滯（連續達標僅提示一次） =====
bool is15mChart = timeframe.period == "15"
bool isWeakGreen = is15mChart and histFast > 0 and histFast <= stallHistLimit
bool isWeakRed   = is15mChart and histFast < 0 and histFast >= -stallHistLimit
bool wasWeakGreen = is15mChart and histFast[1] > 0 and histFast[1] <= stallHistLimit
bool wasWeakRed   = is15mChart and histFast[1] < 0 and histFast[1] >= -stallHistLimit
var int weakGreenBars = 0
var int weakRedBars   = 0
var bool weakGreenSignaled = false
var bool weakRedSignaled   = false

if isWeakGreen
    if wasWeakGreen and histFast > histFast[1]
        weakGreenBars := 0
        weakGreenSignaled := false
    else
        weakGreenBars += 1
else
    weakGreenBars := 0
    weakGreenSignaled := false

if isWeakRed
    if wasWeakRed and histFast < histFast[1]
        weakRedBars := 0
        weakRedSignaled := false
    else
        weakRedBars += 1
else
    weakRedBars := 0
    weakRedSignaled := false

bool stallBearSignal = enableStallSignal and weakGreenBars == stallBars and not weakGreenSignaled and barstate.isconfirmed
bool stallBullSignal = enableStallSignal and weakRedBars == stallBars and not weakRedSignaled and barstate.isconfirmed

if stallBearSignal
    weakGreenSignaled := true

f_analyze_heap(bool enable, bool is_bull_heap, float histFast, float histSlow, int min_bars,
               float major_threshold, float minor_threshold,
               color box_bg_color, color box_border_color, color box_text_color,
               string box_text, bool ema_slope_up, float y_label) =>
    var HeapState state = HeapState.new(na, na, na, na, na, false, na, na)
    bool condition_met = false

    if not enable
        if not na(state.active_box)
            box.delete(state.active_box)
            state.active_box := na
        if not na(state.active_label)
            label.delete(state.active_label)
            state.active_label := na
        state.in_heap          := false
        state.prev_fast_extreme := na
        state.prev_slow_extreme := na
    else
        bool current_in_heap = is_bull_heap ? (histFast > 0) : (histFast < 0)
        if current_in_heap
            if not state.in_heap
                state.in_heap := true
                state.curr_fast_extreme := histFast
                state.curr_slow_extreme := histSlow
                state.start_idx := bar_index
                state.active_box := na
                state.active_label := na
            else
                if is_bull_heap
                    state.curr_fast_extreme := math.max(state.curr_fast_extreme, histFast)
                    state.curr_slow_extreme := math.max(state.curr_slow_extreme, histSlow)
                else
                    state.curr_fast_extreme := math.min(state.curr_fast_extreme, histFast)
                    state.curr_slow_extreme := math.min(state.curr_slow_extreme, histSlow)
        else
            if state.in_heap
                int heap_bars = bar_index - state.start_idx
                if heap_bars >= min_bars
                    state.prev_fast_extreme := state.curr_fast_extreme
                    state.prev_slow_extreme := state.curr_slow_extreme
                state.in_heap := false
                if not na(state.active_box)
                    box.delete(state.active_box)
                    state.active_box := na
                if not na(state.active_label)
                    label.delete(state.active_label)
                    state.active_label := na

        if state.in_heap and not na(state.prev_fast_extreme)
            int bars_in_heap = bar_index - state.start_idx + 1
            bool cond_major     = is_bull_heap ? (state.prev_slow_extreme >= major_threshold) : (math.abs(state.prev_slow_extreme) >= major_threshold)
            bool cond_min_bars  = bars_in_heap >= min_bars
            bool cond_fast_limit = is_bull_heap ? (histFast <= minor_threshold) : (math.abs(histFast) <= minor_threshold)
            bool cond_divergence = is_bull_heap ? (state.curr_fast_extreme < state.prev_fast_extreme) : (math.abs(state.curr_fast_extreme) < math.abs(state.prev_fast_extreme))
            bool cond_no_dark   = is_bull_heap ? ((histSlow <= 0) or (histSlow <= histFast + heapDarkTolerance)) : ((histSlow >= 0) or (math.abs(histSlow) <= math.abs(histFast) + heapDarkTolerance))
            bool cond_ema_slope = is_bull_heap ? ema_slope_up : not ema_slope_up

            if cond_major and cond_min_bars and cond_fast_limit and cond_divergence and cond_no_dark and cond_ema_slope
                condition_met := true
                if na(state.active_box)
                    if is_bull_heap
                        state.active_box := box.new(left = state.start_idx, top = state.curr_fast_extreme, right = bar_index, bottom = 0,
                                                     border_color = box_border_color, border_width = 2, border_style = line.style_solid,
                                                     bgcolor = box_bg_color)
                        state.active_label := label.new(x = math.floor((state.start_idx + bar_index) / 2), y = -y_label, text = box_text,
                                                         color = color.new(color.white, 100), textcolor = box_text_color,
                                                         style = label.style_none, size = size.normal)
                    else
                        state.active_box := box.new(left = state.start_idx, top = 0, right = bar_index, bottom = state.curr_fast_extreme,
                                                     border_color = box_border_color, border_width = 2, border_style = line.style_solid,
                                                     bgcolor = box_bg_color)
                        state.active_label := label.new(x = math.floor((state.start_idx + bar_index) / 2), y = y_label, text = box_text,
                                                         color = color.new(color.white, 100), textcolor = box_text_color,
                                                         style = label.style_none, size = size.normal)
                else
                    box.set_right(state.active_box, bar_index)
                    if is_bull_heap
                        box.set_top(state.active_box, state.curr_fast_extreme)
                    else
                        box.set_bottom(state.active_box, state.curr_fast_extreme)
                    label.set_x(state.active_label, math.floor((state.start_idx + bar_index) / 2))
    [condition_met, state]

bool ema20_rising = ema20 > ema20[1]

float highest_abs_hist_for_label = ta.highest(math.abs(histFast), length = 50)
float y_label_pos = highest_abs_hist_for_label * 1.5
