// --- [OB 優化] 共用 FVG 建立函數，消除多空重複代碼 ---
createFvgEntry(bool isBull, float fvg_upper, float fvg_lower, array<float> cls_arr, array<float> vol_arr, color fvg_color) =>
    fvg_box = box.new(bar_index - 1, fvg_upper, bar_index + 3, fvg_lower, border_color = na, border_width = 0, bgcolor = color.new(fvg_color, 70), tooltip = "FVG (Fair Value Gap)\nVP 為 FVG 形成當下的成交量分佈快照，非即時更新。\n顏色：粉紫=空方 FVG | 綠=多方 FVG")  // [Bug #12 修復] tooltip 註明 VP 為快照
    binSize = (fvg_upper - fvg_lower) / bins

    // [OD 優化] 只在 FVG 觸發時分配陣列
    volumeP     = array.new<float>(bins, 0.)
    volumeBoxes = array.new<box>(bins, box(na))
    poc_line    = line(na)
    poc_lbl     = label(na)

    // [OA 優化] VP 計算迴圈反轉：先遍歷 K 棒一次 O(candles)，取代原本 O(bins × candles)
    // [Bug #3 修正] 1m 圖 (hasLowerData=false) 跳過 VP 計算；[New#C] showVP=false 也跳過 (節省 box 配額)
    if hasLowerData and showVP and cls_arr.size() > 1
        for i = 0 to cls_arr.size() - 1
            cls = cls_arr.get(i)
            v_val = vol_arr.get(i)
            if cls >= fvg_lower and cls <= fvg_upper
                binIdx = math.floor((cls - fvg_lower) / binSize)
                binIdx := math.max(0, math.min(binIdx, bins - 1))
                volumeP.set(binIdx, volumeP.get(binIdx) + v_val)

    // 繪製 VP Boxes + 找 POC
    // [Bug #3 修正] 1m 圖完全跳過；[New#C] showVP=false 也不創建 box
    if hasLowerData and showVP
        float maxVol = volumeP.max()
        bool pocFound = false
        for k = 0 to bins - 1
            lower = fvg_lower + binSize * k
            upper = lower + binSize
            mid   = lower + binSize / 2
            volumeBoxes.set(k, box.new(bar_index - 1, upper, bar_index - 1, lower, border_width = 0))
            v_val_bin = volumeP.get(k)
            if maxVol != 0 and v_val_bin == maxVol and not pocFound
                poc_line := line.new(bar_index - 1, mid, bar_index - 1, mid, color = fvg_color)
                // [P11 修復] 成交量一律正值，多空用前綴區分
                string volText = isBull ? str.tostring(maxVol, format.volume) : "▼ " + str.tostring(maxVol, format.volume)
                poc_lbl  := label.new(bar_index - 1, mid, volText, style = label.style_label_left, color = fvg_color, textcolor = chart.fg_color, size = size.tiny)
                pocFound := true

    fvg.new(fvg_box, isBull, volumeP, volumeBoxes, poc_line, poc_lbl)

detectFairValueGap() =>
    var fvgs = array.new<fvg>()

    bool bullish_gap_condition = low > high[2] and high[1] > high[2] and bullish_gap_size > gap_filter
    bool bearish_gap_condition = high < low[2] and low[1] < low[2] and bearish_gap_size > gap_filter

    if c_lower.size() == 0 or v_lower.size() == 0
        bullish_gap_condition := false
        bearish_gap_condition := false

    cls_array = c_lower[1]
    vol_array = v_lower[1]

    // --- 建立新 FVG ---
    if bullish_gap_condition
        fvgs.push(createFvgEntry(true, low, high[2], cls_array, vol_array, bull_color))

    if bearish_gap_condition
        fvgs.push(createFvgEntry(false, low[2], high, cls_array, vol_array, bear_color))

    // --- 超過上限時清理最舊 ---
    if fvgs.size() > 10
        f = fvgs.shift()
        f.FVG.delete()
        f.poc.delete()
        f.pocLbl.delete()
        if hasLowerData and showVP
            for b in f.volumeBoxes
                b.delete()

    // --- [P3 修復] 反向迭代安全刪除 + 更新繪圖 ---
    if fvgs.size() > 0
        for i = fvgs.size() - 1 to 0 by 1
            f = fvgs.get(i)
            f_left = f.FVG.get_left()

            // 超齡清理
            if (bar_index - f_left) > 950
                f.FVG.delete()
                if hasLowerData and showVP
                    for b in f.volumeBoxes
                        b.delete()
                f.poc.delete()
                f.pocLbl.delete()
                fvgs.remove(i)
                continue

            f.FVG.set_right(bar_index + fvg_extend)  // [Bug #11 修復] 25 改為可配置
            top   = f.FVG.get_top()
            bot   = f.FVG.get_bottom()
            left  = f.FVG.get_left()
            right = f.FVG.get_right()
            binSize = (top - bot) / bins
            boxes   = f.volumeBoxes
            f.poc.set_x2(right)
            f.pocLbl.set_x(right)
            // [Bug #3 修正] 1m 圖資料樣本少（單根 K），加 ⚠️ 警告標示，避免誤判 POC
            float vpSum = f.volumeProfile.sum()
            string vpText = str.tostring(vpSum, format.volume)
            f.FVG.set_text(not hasLowerData ? "⚠ 1m " + vpText : vpText)
            f.FVG.set_text_halign(text.align_right)
            f.FVG.set_text_valign(f.isBull ? text.align_bottom : text.align_top)
            f.FVG.set_text_size(size.tiny)
            f.FVG.set_text_color(chart.fg_color)

            // 價格穿越 FVG 時刪除
            bool shouldRemove = f.isBull ? (low < f.FVG.get_bottom()) : (high > f.FVG.get_top())
            if shouldRemove
                f.FVG.delete()
                if hasLowerData and showVP  // [New#C 修復] VP box 未創建時不迭代刪除
                    for b in boxes
                        b.delete()
                f.poc.delete()
                f.pocLbl.delete()
                fvgs.remove(i)
                continue

            // [OC 優化] VP Box 寬度統一公式
            // [New#C 修復] showVP=false 或 1m 圖跳過整個 VP box 更新 (避免 na.set_right 報錯)
            if hasLowerData and showVP
                float volSum = f.volumeProfile.max()
                if volSum != 0
                    for k = 0 to bins - 1
                        lower = bot + binSize * k
                        volRatio = f.volumeProfile.get(k) / volSum
                        boxWidth = math.round(volRatio * (right - left) / 2)
                        box_id   = boxes.get(k)
                        fvg_color = f.isBull ? bull_color : bear_color
                        bgColor  = color.from_gradient(f.volumeProfile.get(k), 0, volSum, color.new(fvg_color, 100), fvg_color)
                        if boxWidth != 0
                            box_id.set_right(left + boxWidth)
                            box_id.set_bgcolor(bgColor)

detectFairValueGap()

// 📌 [P8 修復] draw_level 參數型別補齊

method draw_level(float src, int n, int w, color col) =>
    line.new(x1 = n, y1 = src, x2 = n, y2 = src, style = line.style_solid, width = math.round(w / 3), color = col)

// --- [P6 修復] check_cross_levels 反向迭代版 ---
// --- [Bug #4 修復] 同步刪除對應 label；[Bug #7 修復] age > 300 也清理 ---
check_cross_levels(arrayOfLines) =>
    if array.size(arrayOfLines) > 0
        for lineNo = array.size(arrayOfLines) - 1 to 0 by 1
            line currentLine = array.get(arrayOfLines, lineNo)
            float lineLevel = line.get_price(currentLine, bar_index)
            bool lineWasCrossed = math.sign(close[1] - lineLevel) != math.sign(close - lineLevel)
            // [Bug #7 修復] 樞紐點年齡 (current_bar - pivot_bar) > 300 則清理
            bool isOld = bar_index - line.get_x1(currentLine) > 300
            if lineWasCrossed or isOld
                line.set_style(currentLine, line.style_dotted)
                if array.size(pivot_labels) > lineNo
                    label.delete(array.get(pivot_labels, lineNo))
                    array.remove(pivot_labels, lineNo)
                array.remove(arrayOfLines, lineNo)
                int(na)
            else
                line.set_x2(currentLine, bar_index)
                int(na)

// =====================================================================================================================}

width2 := (not na(pl) and norm_vol > filter_vol and barstate.isconfirmed) ? math.min(11, math.ceil(norm_vol)) : 0
if not na(pl) and norm_vol > filter_vol and barstate.isconfirmed
    int n = bar_index - rightBars
    float src = low[rightBars]
    chart.point cp = chart.point.from_index(n, src)
    if levels.size() >= maxLevelLines  // [Bug #5 修復] 上限保護
        line.delete(array.shift(levels))
        if array.size(pivot_labels) > 0
            label.delete(array.shift(pivot_labels))
    levels.push(src.draw_level(n, width2, lower_col))

    pivot_labels.push(newLbl)

    int current_minute_of_day = hour(time, "Asia/Taipei") * 60 + minute(time, "Asia/Taipei")
    int minutes_to_0845 = current_minute_of_day - (8 * 60 + 45)

    line.new(x1 = bar_index - bars_back, y1 = orb_h,
             x2 = bar_index + orb_length, y2 = orb_h,
             color = orb_color, width = orb_width)
    line.new(x1 = bar_index - bars_back, y1 = orb_l,
