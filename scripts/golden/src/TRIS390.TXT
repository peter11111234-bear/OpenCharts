f_supertrend(period, multiplier) =>
    var float up = na
    var float dn = na
    var float trend = na
    if barstate.isfirst
        up := 0.0
        dn := 0.0
        trend := 1.0
    float atr = ta.atr(period)
    float hl2_price = (high + low) / 2
    float upperBand = hl2_price + (multiplier * atr)
    float lowerBand = hl2_price - (multiplier * atr)
    float prev_up = nz(up[1], lowerBand)
    float prev_dn = nz(dn[1], upperBand)
    up := close[1] > prev_up ? math.max(lowerBand, prev_up) : lowerBand
    dn := close[1] < prev_dn ? math.min(upperBand, prev_dn) : upperBand
    trend := (nz(trend[1], 1.0) == -1.0 and close > prev_dn) ? 1.0 :
             (nz(trend[1], 1.0) == 1.0 and close < prev_up) ? -1.0 : nz(trend[1], 1.0)
    float stLine = trend == 1.0 ? up : dn
    [stLine, int(trend)]

f_rangefilter(src, period, multiplier) =>
    var float filt = na
    if barstate.isfirst
        filt := src
    float wper = period * 2 - 1
    float avrng = ta.ema(math.abs(src - src[1]), period)
    float smoothrng = ta.ema(avrng, int(wper)) * multiplier
    filt := na(filt[1]) ? src : src > nz(filt[1]) ? ((src - smoothrng) < nz(filt[1]) ? nz(filt[1]) : (src - smoothrng)) :
             ((src + smoothrng) > nz(filt[1]) ? nz(filt[1]) : (src + smoothrng))
    var int upward = 0
    var int downward = 0
    upward := filt > filt[1] ? nz(upward[1]) + 1 : filt < filt[1] ? 0 : nz(upward[1])
    downward := filt < filt[1] ? nz(downward[1]) + 1 : filt > filt[1] ? 0 : nz(downward[1])
    int trend = upward > 0 ? 1 : downward > 0 ? -1 : 0
    [filt, int(trend)]

f_halftrend(amplitude, deviation, timeoutBars) =>
    var int htState = 0
    var int stateBars = 0
    if barstate.isfirst
        htState := 0
        stateBars := 0
    float atrValue = ta.atr(100)
    float maxHigh = ta.highest(high, amplitude)
    float minLow = ta.lowest(low, amplitude)
    float changeAmplitude = ta.change(close, amplitude)
    bool buyCond = low > nz(minLow[1]) and close > close[amplitude] and changeAmplitude > atrValue * deviation
    bool sellCond = high < nz(maxHigh[1]) and close < close[amplitude] and changeAmplitude < -atrValue * deviation
    if buyCond
        htState := 1
        stateBars := 0
    else if sellCond
        htState := -1
        stateBars := 0
    else
        stateBars := nz(stateBars[1], 0) + 1
        if timeoutBars > 0 and stateBars > timeoutBars
            htState := 0
            stateBars := 0
    bool upTrend = htState == 1
    bool downTrend = htState == -1
    bool buySignal = upTrend and not upTrend[1]
    bool sellSignal = downTrend and not downTrend[1]
    [buySignal, sellSignal, upTrend, downTrend]

f_calc_score(st, rf, up, dn) =>
    float stDir = st == 1 ? 1 : st == -1 ? -1 : 0
    float rfDir = rf == 1 ? 1 : rf == -1 ? -1 : 0
    float htDir = up ? 1 : dn ? -1 : 0
    float bullCount = (stDir == 1 ? 1 : 0) + (rfDir == 1 ? 1 : 0) + (htDir == 1 ? 1 : 0)
    float bearCount = (stDir == -1 ? 1 : 0) + (rfDir == -1 ? 1 : 0) + (htDir == -1 ? 1 : 0)
    float s = 0.0
    s += (st == 1 ? 0.4 : st == -1 ? -0.4 : 0)
    s += (rf == 1 ? 0.3 : rf == -1 ? -0.3 : 0)
    s += (up ? 0.3 : dn ? -0.3 : 0)
    bool consensus = bullCount >= 2 or bearCount >= 2
    float finalScore = consensus ? s : s * 0.5
    math.max(-1.0, math.min(1.0, finalScore))

float fastScoreMonthly = f_calc_score(stM, rfM, htUpM, htDnM)
float fastScoreWeekly  = f_calc_score(stW, rfW, htUpW, htDnW)
float fastScoreDaily   = f_calc_score(stD, rfD, htUpD, htDnD)
float fastScore4H      = f_calc_score(st4H, rf4H, htUp4H, htDn4H)
float fastScore1H      = f_calc_score(st1H, rf1H, htUp1H, htDn1H)
float fastScore30M     = f_calc_score(st30, rf30, htUp30, htDn30)
float fastScore15M     = f_calc_score(st15, rf15, htUp15, htDn15)
float fastScore5M      = f_calc_score(st5, rf5, htUp5, htDn5)
float fastScoreExec    = f_calc_score(stE, rfE, htUpE, htDnE)

float fastTotalScore = (fastScore4H * w4H) + (fastScore1H * w1H) + (fastScoreExec * wExec)

float rLow_fb  = not na(barsPerHTF) ? ta.lowest(low, barsPerHTF) : na
float rClose_fb = close
float pOpen_fb = not na(barsPerHTF) ? open[barsPerHTF * 2 - 1] : na
float pHigh_fb = not na(barsPerHTF) ? ta.highest(high[barsPerHTF], barsPerHTF) : na
float pLow_fb  = not na(barsPerHTF) ? ta.lowest(low[barsPerHTF], barsPerHTF) : na
float pClose_fb = not na(barsPerHTF) ? close[barsPerHTF] : na

// ── 30分單12H：錨定 15:00 動態會話K ──
var float s720_O = na
var float s720_H = na
var float s720_L = na
bool is15Start = hour == 15 and minute == 0
if is15Start and barstate.isnew
    s720_O := open
    s720_H := high
    s720_L := low
else if na(s720_O)
    s720_O := open
    s720_H := high
    s720_L := low
else
    s720_H := math.max(s720_H, high)
    s720_L := math.min(s720_L, low)
float s720_C = close

// ── 60分/1H單日K：已停用（v3.87 移除，1H 圖不再繪製單日K） ──
bool useSecurity = not na(targetTF) and targetTF != "720" and targetTF != "D"
float rollOpen_base  = not na(targetTF) ? (useSecurity ? (na(oHTF) ? rOpen_fb : oHTF) : rOpen_fb) : na
float rollHigh_base  = not na(targetTF) ? (useSecurity ? (na(hHTF) ? rHigh_fb : hHTF) : rHigh_fb) : na
float rollLow_base   = not na(targetTF) ? (useSecurity ? (na(lHTF) ? rLow_fb : lHTF) : rLow_fb) : na
float rollClose_base = not na(targetTF) ? (useSecurity ? (na(cHTF) ? rClose_fb : cHTF) : rClose_fb) : na
float rollOpen  = targetTF == "720" ? s720_O : rollOpen_base
float rollHigh  = targetTF == "720" ? s720_H : rollHigh_base
float rollLow   = targetTF == "720" ? s720_L : rollLow_base
float rollClose = targetTF == "720" ? s720_C : rollClose_base
float prevFrozen_O = not na(targetTF) ? (useSecurity ? (na(oHTF1) ? pOpen_fb : oHTF1) : pOpen_fb) : na
float prevFrozen_H = not na(targetTF) ? (useSecurity ? (na(hHTF1) ? pHigh_fb : hHTF1) : pHigh_fb) : na
float prevFrozen_L = not na(targetTF) ? (useSecurity ? (na(lHTF1) ? pLow_fb : lHTF1) : pLow_fb) : na
float prevFrozen_C = not na(targetTF) ? (useSecurity ? (na(cHTF1) ? pClose_fb : cHTF1) : pClose_fb) : na

// 繪圖生命週期管理
var label rollLabel = na
var line rollWickLine = na
var box rollBodyBox = na
var line prevWickLine = na
var box prevBodyBox = na

if barstate.islast
    label.delete(rollLabel)
    line.delete(rollWickLine)
    box.delete(rollBodyBox)
    line.delete(prevWickLine)
    box.delete(prevBodyBox)

// ── 自適應雙K繪圖（xloc.bar_time 時間座標） ──
float atrForBody = ta.atr(14)
bool rollReady = not na(targetTF) and not na(rollOpen)
bool prevReady = not isSingleMode and rollReady
if barstate.islast and showRoll and rollReady
    bool isBull = rollClose >= rollOpen
    color clr = isBull ? color.green : color.red
    color clrFill = isBull ? color.new(color.green, 50) : color.new(color.red, 50)

    float pO = not na(prevFrozen_O) ? prevFrozen_O : not na(pOpen_fb) ? pOpen_fb : rollOpen
    float pH = not na(prevFrozen_H) ? prevFrozen_H : not na(pHigh_fb) ? pHigh_fb : rollHigh
    float pL = not na(prevFrozen_L) ? prevFrozen_L : not na(pLow_fb) ? pLow_fb : rollLow
    float pC = not na(prevFrozen_C) ? prevFrozen_C : not na(pClose_fb) ? pClose_fb : rollClose

    // bxTime = 前一小時K位置(左)，當前K在其右側
    int curBxTime = bxTime + boxWTime + gapTime
    int wickXTime = curBxTime + boxWTime / 2
    rollWickLine := line.new(wickXTime, rollHigh, wickXTime, rollLow, color=color.gray, width=1, xloc=xloc.bar_time)

    float bodyTop = math.max(rollOpen, rollClose)
    float bodyBottom = math.min(rollOpen, rollClose)
    float bodyH = bodyTop - bodyBottom
    float minBody = na(atrForBody) ? 0 : atrForBody * 0.04
    if bodyH < minBody
        float mid = (bodyTop + bodyBottom) / 2
        bodyTop := mid + minBody / 2
        bodyBottom := mid - minBody / 2
    rollBodyBox := box.new(curBxTime, bodyTop, curBxTime + boxWTime, bodyBottom,
                           bgcolor=clrFill, border_color=clr, border_width=1, xloc=xloc.bar_time)

    int labelTime = curBxTime + boxWTime + 2 * secPerBar_rt * 1000
    rollLabel := label.new(labelTime, rollHigh,
                           targetLabel + "\n" +
                           "O:" + str.tostring(rollOpen, "#.##") + "\n" +
                           "H:" + str.tostring(rollHigh, "#.##") + "\n" +
                           "L:" + str.tostring(rollLow, "#.##") + "\n" +
                           "C:" + str.tostring(rollClose, "#.##"),
                           style=label.style_label_left,
                           color=color.new(#222222, 20), textcolor=color.white,
                           textalign=text.align_left, size=size.small, xloc=xloc.bar_time)

    if prevReady
        int prevBxTime = bxTime

        bool prevBull = pC >= pO
        color prevClr = prevBull ? color.green : color.red
        color prevFill = prevBull ? color.new(color.green, 50) : color.new(color.red, 50)

        int prevWickXTime = prevBxTime + boxWTime / 2
        prevWickLine := line.new(prevWickXTime, pH, prevWickXTime, pL, color=color.gray, width=1, xloc=xloc.bar_time)

        float pBodyTop = math.max(pO, pC)
        float pBodyBot = math.min(pO, pC)
        float pBodyH = pBodyTop - pBodyBot
        if pBodyH < minBody
            float pMid = (pBodyTop + pBodyBot) / 2
            pBodyTop := pMid + minBody / 2
            pBodyBot := pMid - minBody / 2
        prevBodyBox := box.new(prevBxTime, pBodyTop, prevBxTime + boxWTime, pBodyBot,
                               bgcolor=prevFill, border_color=prevClr, border_width=1, xloc=xloc.bar_time)
