// [第一階段] 定義循環緩衝區的 UDT
type CircularBuffer
    float[] values = na
    int[]   bars   = na
    int[]   dirs   = na
    int     head   = -1
    int     count  = 0
    int     capacity = 0

method initBuffer(CircularBuffer self, int cap) =>
    if cap <= 0
        runtime.error("CircularBuffer capacity must be > 0")
    if self.capacity != cap
        self.capacity := cap
        self.values := array.new_float(cap, na)
        self.bars   := array.new_int(cap, na)
        self.dirs   := array.new_int(cap, na)
        self.head   := -1
        self.count  := 0

method push(CircularBuffer self, float val, int bar, int dir) =>
    if self.capacity > 0
        self.head := (self.head + 1) % self.capacity
        array.set(self.values, self.head, val)
        array.set(self.bars, self.head, bar)
        array.set(self.dirs, self.head, dir)
        if self.count < self.capacity
            self.count += 1

method get(CircularBuffer self, int n) =>
    if n < 0 or n >= self.count
        na
    else
        int idx = (self.head - n + self.capacity) % self.capacity
        array.get(self.values, idx)

method getBar(CircularBuffer self, int n) =>
    if n < 0 or n >= self.count
        na
    else
        int idx = (self.head - n + self.capacity) % self.capacity
        array.get(self.bars, idx)

method getDir(CircularBuffer self, int n) =>
    if n < 0 or n >= self.count
        na
    else
        int idx = (self.head - n + self.capacity) % self.capacity
        array.get(self.dirs, idx)

method pop(CircularBuffer self) =>
    if self.count > 0
        self.head := (self.head - 1 + self.capacity) % self.capacity
        self.count -= 1

method updateTop(CircularBuffer self, float val, int bar, int dir) =>
    if self.count > 0
        array.set(self.values, self.head, val)
        array.set(self.bars, self.head, bar)
        array.set(self.dirs, self.head, dir)

// [第二階段] 繪圖物件生命週期集中管理
type DrawObj
    line   ln = na
    label  lbl = na
    box    bx = na
    int    barCreated

type DrawManager
    DrawObj[] objects
    int maxObjects
    int maxBarsKeep
    int headIndex = 0  // O(1) Ring Buffer 寫入指標，取代 array.shift()

method addLine(DrawManager dm, line l) =>
    if array.size(dm.objects) < dm.maxObjects
        array.push(dm.objects, DrawObj.new(ln = l, barCreated = bar_index))
    else
        DrawObj old = array.get(dm.objects, dm.headIndex)
        if not na(old.ln)
            line.delete(old.ln)
        if not na(old.lbl)
            label.delete(old.lbl)
        if not na(old.bx)
            box.delete(old.bx)
        array.set(dm.objects, dm.headIndex, DrawObj.new(ln = l, barCreated = bar_index))
        dm.headIndex := (dm.headIndex + 1) % dm.maxObjects
    l

method addLabel(DrawManager dm, label lbl) =>
    if array.size(dm.objects) < dm.maxObjects
        array.push(dm.objects, DrawObj.new(lbl = lbl, barCreated = bar_index))
    else
        DrawObj old = array.get(dm.objects, dm.headIndex)
        if not na(old.ln)
            line.delete(old.ln)
        if not na(old.lbl)
            label.delete(old.lbl)
        if not na(old.bx)
            box.delete(old.bx)
        array.set(dm.objects, dm.headIndex, DrawObj.new(lbl = lbl, barCreated = bar_index))
        dm.headIndex := (dm.headIndex + 1) % dm.maxObjects
    lbl

method addBox(DrawManager dm, box bx) =>
    if array.size(dm.objects) < dm.maxObjects
        array.push(dm.objects, DrawObj.new(bx = bx, barCreated = bar_index))
    else
        DrawObj old = array.get(dm.objects, dm.headIndex)
        if not na(old.ln)
            line.delete(old.ln)
        if not na(old.lbl)
            label.delete(old.lbl)
        if not na(old.bx)
            box.delete(old.bx)
        array.set(dm.objects, dm.headIndex, DrawObj.new(bx = bx, barCreated = bar_index))
        dm.headIndex := (dm.headIndex + 1) % dm.maxObjects
    bx

method cleanup(DrawManager dm) =>
    if array.size(dm.objects) > 0
        for i = 0 to array.size(dm.objects) - 1
            DrawObj obj = array.get(dm.objects, i)
            bool has_content = not na(obj.ln) or not na(obj.lbl) or not na(obj.bx)
            if has_content and bar_index - obj.barCreated > dm.maxBarsKeep
                if not na(obj.ln)
                    line.delete(obj.ln)
                if not na(obj.lbl)
                    label.delete(obj.lbl)
                if not na(obj.bx)
                    box.delete(obj.bx)
                // O(1) 原址清空：刪除繪圖物件但保留陣列槽位，維持 Ring Buffer 固定長度
                array.set(dm.objects, i, DrawObj.new(barCreated = bar_index))

    array<string> names = array.from("N: ", "E: ", "V: ")
    array<float> vals = array.from(valN, valE, valV)
    array<bool> dones = array.from(dN, dE, dV)
    for i = 0 to 1
        for j = i + 1 to 2
            if array.get(vals, j) > array.get(vals, i)
                float tmpV = array.get(vals, i)
                array.set(vals, i, array.get(vals, j))
                array.set(vals, j, tmpV)
                string tmpN = array.get(names, i)
                array.set(names, i, array.get(names, j))
                array.set(names, j, tmpN)
                bool tmpD = array.get(dones, i)
                array.set(dones, i, array.get(dones, j))
                array.set(dones, j, tmpD)
    string res = title + "\n"
    for i = 0 to 2
        res += array.get(names, i) + str.tostring(array.get(vals, i), "#.0") + (array.get(dones, i) ? " ✓" : "") + (i < 2 ? "\n" : "")
    res

harmonic_pivots(length) =>
    float phigh = ta.highestbars(high, length) == 0 ? high : na
    float plow = ta.lowestbars(low, length) == 0 ? low : na
    dir = 0
    dir := not na(plow) and na(phigh) ? -1 : dir[1]
    dir := not na(phigh) and na(plow) ? 1 : dir
    [dir, phigh, plow]

harmonic_zigzag(length, CircularBuffer cb) =>
    [dir, phigh, plow] = harmonic_pivots(length)
    dirchanged = ta.change(dir)
    if not na(phigh) or not na(plow)
        value = dir == 1 ? phigh : plow
        bar = bar_index
        newDir = dir
        if dirchanged == 0 and cb.count >= 1
            pivot = cb.get(0)
            pivotbar = cb.getBar(0)
            pivotdir = cb.getDir(0)
            cb.pop()
            useNewValues = value * pivotdir < pivot * pivotdir
            value := useNewValues ? pivot : value
            bar := useNewValues ? pivotbar : bar
        if cb.count >= 2
            lastPoint = cb.get(1)
            newDir := dir * value > dir * lastPoint ? dir * 2 : dir
        cb.push(value, bar, newDir)

        lowPoint = math.min(x, a, b, c, d)

// 正向常規/隱藏背離檢測 — 中點抽樣降維 O(P×L) → O(P)
div_pos_reg_hid(src, cond) =>
    divlen = 0
    prsc = div_source == "Close" ? close : low
    if div_trigger and (div_dontconfirm or src > src[1] or close > close[1])
        sp = div_dontconfirm ? 0 : 1
        for x = 0 to div_maxpp - 1
            int barAtX = div_pl_cb.getBar(x)
            _detectLen = bar_index - barAtX + div_prd
            if na(barAtX) or barAtX == 0 or _detectLen > div_maxbars
                break
            if _detectLen > 5 and
               ((cond == 1 and src[sp] > src[_detectLen] and prsc[sp] < nz(div_pl_cb.get(x))) or
               (cond == 2 and src[sp] < src[_detectLen] and prsc[sp] > nz(div_pl_cb.get(x))))
                slope1 = (src[sp] - src[_detectLen]) / (_detectLen - sp)
                slope2 = (close[sp] - close[_detectLen]) / (_detectLen - sp)
                // 中點抽樣：僅校驗 25%/50%/75% 三個關鍵點，取代逐根掃描
                rangeLen = _detectLen - sp
                int mid1 = sp + math.max(1, math.round(rangeLen * 0.25))
                int mid2 = sp + math.max(1, math.round(rangeLen * 0.50))
                int mid3 = sp + math.max(1, math.round(rangeLen * 0.75))
                float line_src_m1 = src[sp] - slope1 * (mid1 - sp)
                float line_cls_m1 = close[sp] - slope2 * (mid1 - sp)
                float line_src_m2 = src[sp] - slope1 * (mid2 - sp)
                float line_cls_m2 = close[sp] - slope2 * (mid2 - sp)
                float line_src_m3 = src[sp] - slope1 * (mid3 - sp)
                float line_cls_m3 = close[sp] - slope2 * (mid3 - sp)
                arrived = true
                if src[mid1] < line_src_m1 or nz(close[mid1]) < line_cls_m1
                    arrived := false
                if arrived and (src[mid2] < line_src_m2 or nz(close[mid2]) < line_cls_m2)
                    arrived := false
                if arrived and (src[mid3] < line_src_m3 or nz(close[mid3]) < line_cls_m3)
                    arrived := false
                if arrived
                    divlen := _detectLen
                    break
    divlen

        array.push(smc.fvg_is_bull, true)
        while array.size(smc.fvg_boxes) > 40
            box.delete(array.shift(smc.fvg_boxes))
            array.shift(smc.fvg_is_bull)

    float ms_bear_gap_top = low[2]
    float ms_bear_gap_btm = high
    float ms_bear_size    = ms_bear_gap_top - ms_bear_gap_btm
    bool ms_pass_bear_trend = ms_filterTrend ? (smc.os == 0) : true
    
    if ms_bear_gap_btm < ms_bear_gap_top and ms_bear_size >= ms_min_gap_size and ms_pass_bear_trend
        box fvg_s = box.new(left = bar_index - 2, top = ms_bear_gap_top, right = bar_index - 2 + ms_fvgLength, bottom = ms_bear_gap_btm, 
             border_color = ms_fvgBearCss, bgcolor = ms_fvgBearCss, extend = extend.none, xloc = xloc.bar_index, 
             text = "FVG", text_size = size.small, text_color = color.new(color.white, 20), 
             text_halign = text.align_right, text_valign = text.align_center)
        array.push(smc.fvg_boxes, fvg_s)
        array.push(smc.fvg_is_bull, false)
        while array.size(smc.fvg_boxes) > 40
            box.delete(array.shift(smc.fvg_boxes))
            array.shift(smc.fvg_is_bull)

    // FVG 填補消除：當蠟燭收盤價完全穿過 FVG 時刪除
    if barstate.isconfirmed and array.size(smc.fvg_boxes) > 0
        for i = array.size(smc.fvg_boxes) - 1 to 0
            box fvg_box = array.get(smc.fvg_boxes, i)
            float fvg_top = box.get_top(fvg_box)
            float fvg_btm = box.get_bottom(fvg_box)
            // Bullish FVG 被空方穿越（收盤跌破底部）→ 消除
            // Bearish FVG 被多方穿越（收盤突破頂部）→ 消除
            bool fvg_is_bull_val = array.get(smc.fvg_is_bull, i)
            bool is_mitigated = fvg_is_bull_val ? (close < fvg_btm) : (close > fvg_top)
            if is_mitigated
                box.delete(fvg_box)
                array.remove(smc.fvg_boxes, i)
                array.remove(smc.fvg_is_bull, i)

bool ms_bos_bull_condition = smc.sbtm_crossed and smc.os == 1 and not na(ms_bos_top_ref)

    if ms_showBos
